const GATEWAY = "http://localhost:8090";
const json = { "Content-Type": "application/json" };

// 1. Sign in - two steps: password, then the emailed one-time code
await fetch(GATEWAY + "/api/auth/login", {
  method: "POST", headers: json, body: JSON.stringify({ email, password }),
});
const { accessToken, refreshToken } = await (await fetch(GATEWAY + "/api/auth/verify-login-otp", {
  method: "POST", headers: json, body: JSON.stringify({ identifier: email, otp }),
})).json();

// 2. Call anything as that user
await fetch(GATEWAY + "/api/auth/me", { headers: { Authorization: "Bearer " + accessToken } });

// 3. Renew before the access token expires (15 minutes)
await fetch(GATEWAY + "/api/auth/refresh", {
  method: "POST", headers: json, body: JSON.stringify({ refreshToken }),
});
